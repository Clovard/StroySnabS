&Вместо("ВыполнитьКомандуПечати")
Процедура Б24_ВместоВыполнитьКомандуПечати(ИмяМенеджераПечати, ИменаМакетов, МассивОбъектов, ВладелецФормы, ПараметрыПечати)

	Если Б24_ЭтоПечатьСчетаСФаксимиле(ИменаМакетов, ИмяМенеджераПечати) Тогда

		ЗаказСсылка = Б24_ЗаказДляПроверкиДаты(МассивОбъектов, ВладелецФормы);

		Если НЕ ЗначениеЗаполнено(ЗаказСсылка) Тогда
			ПоказатьПредупреждение(, "Сначала запишите заказ клиента.");
			Возврат;
		КонецЕсли;

		ТекстОшибки = Б24_СчетВызовСервера.ТекстОшибкиДатыПередСчетомСФаксимиле(ЗаказСсылка);

		Если НЕ ПустаяСтрока(ТекстОшибки) Тогда
			ПоказатьПредупреждение(, ТекстОшибки);
			Возврат;
		КонецЕсли;

	КонецЕсли;

	ПродолжитьВызов(ИмяМенеджераПечати, ИменаМакетов, МассивОбъектов, ВладелецФормы, ПараметрыПечати);

&НаКлиенте
Функция Б24_ЭтоПечатьСчетаСФаксимиле(ИменаМакетов, ИмяМенеджераПечати)

	Текст = НРег(СокрЛП(Строка(ИменаМакетов) + " " + Строка(ИмяМенеджераПечати)));

	Возврат СтрНайти(Текст, "факсимил") > 0
		Или СтрНайти(Текст, "facsimile") > 0
		Или СтрНайти(Текст, "сфаксимиле") > 0;

КонецФункции

&НаКлиенте
Функция Б24_ЗаказДляПроверкиДаты(МассивОбъектов, ВладелецФормы)

	Если ВладелецФормы <> Неопределено Тогда

		Попытка

			Если ЗначениеЗаполнено(ВладелецФормы.Объект.Ссылка) Тогда
				Возврат ВладелецФормы.Объект.Ссылка;
			КонецЕсли;

		Исключение
		КонецПопытки;

	КонецЕсли;

	Если МассивОбъектов = Неопределено Тогда
		Возврат Неопределено;
	КонецЕсли;

	Если ТипЗнч(МассивОбъектов) = Тип("Массив") Тогда

		Если МассивОбъектов.Количество() = 0 Тогда
			Возврат Неопределено;
		КонецЕсли;

		ОбъектПечати = МассивОбъектов[0];

	Иначе
		ОбъектПечати = МассивОбъектов;
	КонецЕсли;

	Возврат Б24_СчетВызовСервера.ЗаказКлиентаПоОбъектуПечати(ОбъектПечати);
