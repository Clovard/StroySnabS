// Обработка нажатия кнопки «Сформировать КП» на форме заказа.
///////////////////////////////////////////////////////////////////////////////////////////////////////

#Область ПрограммныйИнтерфейс

// Вызывается из команды формы Документ.ЗаказКлиента.
Процедура СформироватьКП(Форма) Экспорт
	
	Если Форма = Неопределено Тогда
		ПоказатьПредупреждение(, "Не удалось определить форму документа.");
		Возврат;
	КонецЕсли;
	
	Если НЕ ЗначениеЗаполнено(Форма.Объект.Ссылка) Или Форма.Объект.Ссылка.Пустая() Тогда
		ПоказатьПредупреждение(, "Сначала запишите заказ клиента.");
		Возврат;
	КонецЕсли;
	
	Если Форма.Модифицированность Тогда
		ПоказатьПредупреждение(, "Заказ изменён. Запишите документ перед формированием КП.");
		Возврат;
	КонецЕсли;
	
	Состояние("Формирование коммерческого предложения и отправка в Bitrix24...");
	
	Результат = Б24КП_ВызовСервера.СформироватьКПиОтправитьВBitrix24(Форма.Объект.Ссылка);
	
	Если Результат.PDFСформирован И Результат.ПечатнаяФорма <> Неопределено Тогда
		Результат.ПечатнаяФорма.Показать(Результат.НазваниеПечатнойФормы);
	КонецЕсли;

	// Стадия меняется при формировании КП — обновить форму
	Если Результат.PDFСформирован Тогда
		Б24КП_ОбновитьСтадиюНаФорме(Форма);
	КонецЕсли;
	
	Если Результат.Успех Тогда
		ПоказатьОповещениеПользователя("Сформировать КП", , Результат.ТекстСообщения);
	ИначеЕсли ЗначениеЗаполнено(Результат.ТекстСообщения) Тогда
		ПоказатьПредупреждение(, Результат.ТекстСообщения);
	ИначеЕсли ЗначениеЗаполнено(Результат.ТекстОшибки) Тогда
		ПоказатьПредупреждение(, Результат.ТекстОшибки);
	Иначе
		ПоказатьПредупреждение(, "Не удалось сформировать КП.");
	КонецЕсли;
	
КонецПроцедуры

Процедура Б24КП_ОбновитьСтадиюНаФорме(Форма)

	Если Форма = Неопределено Тогда
		Возврат;
	КонецЕсли;

	Попытка
		Форма.Прочитать();
	Исключение
		Форма.ОбновитьОтображениеДанных();
	КонецПопытки;

КонецПроцедуры

#КонецОбласти
