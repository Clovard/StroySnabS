#Область Б24_ДатаПередПечатьюФаксимиле

&НаКлиенте
&Перед("Подключаемый_ОбработатьКоманду")
Процедура Б24_ПередПодключаемый_ОбработатьКоманду(Команда, ДополнительныеПараметры, СтандартнаяОбработка)

	Если НЕ Б24_ЭтоФаксимилеКоманда(Команда, ДополнительныеПараметры) Тогда
		Возврат;
	КонецЕсли;

	ТекстОшибки = Б24_ТекстОшибкиДатыНаСервере(Объект.Ссылка);

	Если ПустаяСтрока(ТекстОшибки) Тогда
		Возврат;
	КонецЕсли;

	ПоказатьПредупреждение(, ТекстОшибки);
	СтандартнаяОбработка = Ложь;

КонецПроцедуры

&НаКлиенте
Функция Б24_ЭтоФаксимилеКоманда(Команда, ДополнительныеПараметры)

	Части = Новый Массив;

	Если Команда <> Неопределено Тогда
		Попытка
			Части.Добавить(Строка(Команда.Имя));
		Исключение
		КонецПопытки;
		Попытка
			Части.Добавить(Строка(Команда.Заголовок));
		Исключение
		КонецПопытки;
	КонецЕсли;

	Если ДополнительныеПараметры <> Неопределено
		И ТипЗнч(ДополнительныеПараметры) = Тип("Структура")
		И ДополнительныеПараметры.Свойство("ИменаМакетов") Тогда
		Части.Добавить(Строка(ДополнительныеПараметры.ИменаМакетов));
	КонецЕсли;

	Текст = НРег(СтрСоединить(Части, " "));

	Возврат СтрНайти(Текст, "факсимил") > 0
		Или СтрНайти(Текст, "facsimile") > 0;

КонецФункции

&НаСервере
Функция Б24_ТекстОшибкиДатыНаСервере(ЗаказСсылка)
	Возврат Б24_СчетСервер.ТекстОшибкиДатыПередСчетомСФаксимиле(ЗаказСсылка);
КонецФункции
