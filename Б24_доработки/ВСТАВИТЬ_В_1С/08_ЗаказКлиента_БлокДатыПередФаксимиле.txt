#Область Б24_ДатаПередПечатьюФаксимиле

&НаКлиенте
&Перед("Подключаемый_ОбработатьКоманду")
Процедура Б24_ПередПодключаемый_ОбработатьКоманду(Команда, ДополнительныеПараметры, СтандартнаяОбработка)

	Б24_ПроверитьДатуПередКомандойПечати(Команда, ДополнительныеПараметры, СтандартнаяОбработка);

КонецПроцедуры

&НаКлиенте
Процедура Б24_ПроверитьДатуПередКомандойПечати(Команда, ДополнительныеПараметры, СтандартнаяОбработка)

	Если НЕ Б24_ЭтоКомандаПечатиФаксимиле(Команда, ДополнительныеПараметры) Тогда
		Возврат;
	КонецЕсли;

	ТекстОшибки = Б24_ТекстОшибкиДатыПередФаксимилеНаСервере();

	Если ПустаяСтрока(ТекстОшибки) Тогда
		Возврат;
	КонецЕсли;

	ПоказатьПредупреждение(, ТекстОшибки);
	СтандартнаяОбработка = Ложь;

КонецПроцедуры

&НаКлиенте
Функция Б24_ЭтоКомандаПечатиФаксимиле(Команда, ДополнительныеПараметры)

	Текст = Б24_ТекстКомандыПечати(Команда, ДополнительныеПараметры);

	Возврат Б24_ТекстСодержитФаксимиле(Текст);

КонецФункции

&НаКлиенте
Функция Б24_ТекстКомандыПечати(Команда, ДополнительныеПараметры)

	Части = Новый Массив;

	Если Команда <> Неопределено Тогда

		Попытка
			Части.Добавить(Строка(Команда.Имя));
		Исключение
		КонецПопытки;

		Попытка
			Части.Добавить(Строка(Команда.Заголовок));
		Исключение
		КонецПопытки;

		Попытка
			Части.Добавить(Строка(Команда.Подсказка));
		Исключение
		КонецПопытки;

	КонецЕсли;

	Если ДополнительныеПараметры <> Неопределено Тогда

		Если ТипЗнч(ДополнительныеПараметры) = Тип("Структура") Тогда

			Если ДополнительныеПараметры.Свойство("ИменаМакетов") Тогда
				Части.Добавить(Строка(ДополнительныеПараметры.ИменаМакетов));
			КонецЕсли;

			Если ДополнительныеПараметры.Свойство("Идентификатор") Тогда
				Части.Добавить(Строка(ДополнительныеПараметры.Идентификатор));
			КонецЕсли;

		Иначе
			Части.Добавить(Строка(ДополнительныеПараметры));
		КонецЕсли;

	КонецЕсли;

	Возврат СтрСоединить(Части, " ");

КонецФункции

&НаКлиенте
Функция Б24_ТекстСодержитФаксимиле(Текст)

	ТекстНРег = НРег(СокрЛП(Текст));

	Возврат СтрНайти(ТекстНРег, "факсимил") > 0
		Или СтрНайти(ТекстНРег, "facsimile") > 0;

КонецФункции

&НаСервере
Функция Б24_ТекстОшибкиДатыПередФаксимилеНаСервере()

	ЗаказСсылка = Объект.Ссылка;

	Если НЕ ЗначениеЗаполнено(ЗаказСсылка) Тогда
		Возврат "Сначала запишите заказ клиента.";
	КонецЕсли;

	Возврат Б24_СчетСервер.ТекстОшибкиДатыПередСчетомСФаксимиле(ЗаказСсылка);

КонецФункции

#КонецОбласти
