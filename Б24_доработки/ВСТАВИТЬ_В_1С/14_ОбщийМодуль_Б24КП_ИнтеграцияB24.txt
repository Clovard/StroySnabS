// Заказ уже связан со сделкой B24 (регистр Б24_К_ИдентификаторыДел) — независимо от ИНН.
Функция ЗаказСвязанСоСделкойB24(ЗаказКлиента) Экспорт

	НастройкаПодключения = ПолучитьНастройкуПодключенияB24();

	Если НЕ ЗначениеЗаполнено(НастройкаПодключения) Тогда
		Возврат Ложь;
	КонецЕсли;

	Возврат ЗначениеЗаполнено(ПолучитьОбъектСвязиДляТаймЛайн(НастройкаПодключения, ЗаказКлиента));

КонецФункции

Функция ОтправитьКПВСуществующуюСделку(ЗаказКлиента, ПечатнаяФорма, НазваниеПечатнойФормы) Экспорт
	
	Результат = НовыйРезультатB24();
	
	НастройкаПодключения = ПолучитьНастройкуПодключенияB24();
	Если НЕ ЗначениеЗаполнено(НастройкаПодключения) Тогда
		Результат.ТекстОшибки = "Не настроено подключение к Битрикс24 (справочник Б24_К_НастройкиПодключения).";
		Возврат Результат;
	КонецЕсли;
	
	Если НЕ ДоступенФункционалB24(НастройкаПодключения) Тогда
		Результат.ТекстОшибки = "Функционал выгрузки в Битрикс24 недоступен на тарифном плане / в настройках модуля.";
		Возврат Результат;
	КонецЕсли;
	
	ОбъектСвязи = ПолучитьОбъектСвязиДляТаймЛайн(НастройкаПодключения, ЗаказКлиента);
	
	Если НЕ ЗначениеЗаполнено(ОбъектСвязи) Тогда
		Результат.ТекстОшибки =
			"Заказ не связан со сделкой в Битрикс24. При наличии ИНН новая сделка не создаётся — "
			+ "сначала установите связь заказа со сделкой (как для счетов) или откройте заказ в Bitrix24.";
		Возврат Результат;
	КонецЕсли;
	
	Результат.IDСделкиB24 = ПолучитьIDСделкиИзРегистра(НастройкаПодключения, ОбъектСвязи);
	
	МодульRest = ОбщегоНазначения.ОбщийМодуль("Б24_К_RestApiВызовСервера");
	ДанныеВыгрузки = МодульRest.ВыгрузитьПечатнуюФормуВТаймЛайн(
		НастройкаПодключения,
		ОбъектСвязи,
		НазваниеПечатнойФормы,
		ПечатнаяФорма);
	
	Если ДанныеВыгрузки.Выгружено = Истина Тогда
		Результат.Успех = Истина;
	Иначе
		Результат.ТекстОшибки = "Не удалось выгрузить КП в таймлайн сделки Битрикс24 (штатный метод модуля B24).";
	КонецЕсли;
	
	Возврат Результат;
	
КонецФункции

Функция СоздатьСделкуИОтправитьКП(ЗаказКлиента, ПечатнаяФорма, НазваниеПечатнойФормы) Экспорт
	
	Результат = НовыйРезультатB24();
	
	ВебхукURL = СокрЛП(Константы.Б24КП_ВебхукURL.Получить());
	Если НЕ ЗначениеЗаполнено(ВебхукURL) Тогда
		Результат.ТекстОшибки =
			"Не заполнена константа Б24КП_ВебхукURL (входящий вебхук Bitrix24). "
			+ "Нужна для создания сделки, когда у контрагента нет ИНН.";
		Возврат Результат;
	КонецЕсли;
	
	IDВоронки = СокрЛП(Строка(Константы.Б24КП_IDВоронки.Получить()));
	Если НЕ ЗначениеЗаполнено(IDВоронки) Тогда
		Результат.ТекстОшибки =
			"Не заполнена константа Б24КП_IDВоронки (CATEGORY_ID воронки Bitrix24) для сценария без ИНН.";
		Возврат Результат;
	КонецЕсли;
	
	Реквизиты = ОбщегоНазначения.ЗначенияРеквизитовОбъекта(ЗаказКлиента, "Номер, Дата, СуммаДокумента, Валюта");
	
	ЗаголовокСделки = СтрШаблон("КП из 1С — Заказ %1", СокрЛП(Строка(Реквизиты.Номер)));
	
	ПоляСделки = Новый Структура;
	ПоляСделки.Вставить("TITLE", ЗаголовокСделки);
	ПоляСделки.Вставить("CATEGORY_ID", IDВоронки);
	ПоляСделки.Вставить("OPPORTUNITY", Реквизиты.СуммаДокумента);
	ПоляСделки.Вставить("CURRENCY_ID", КодВалютыB24(Реквизиты.Валюта));
	ПоляСделки.Вставить("COMMENTS",
		СтрШаблон("Создано из 1С автоматически (нет ИНН контрагента). Заказ от %1. Компанию/контакт дозаполняет менеджер.",
			Формат(Реквизиты.Дата, "ДЛФ=DD")));
	
	ОтветСделки = ВызватьRestМетод(ВебхукURL, "crm.deal.add", Новый Структура("fields", ПоляСделки));
	
	Если НЕ ОтветСделки.Успех Тогда
		Результат.ТекстОшибки = "Не удалось создать сделку в Bitrix24: " + ОтветСделки.ТекстОшибки;
		Возврат Результат;
	КонецЕсли;
	
	IDСделки = XMLСтрока(ОтветСделки.Результат);
	Если НЕ ЗначениеЗаполнено(IDСделки) Тогда
		Результат.ТекстОшибки = "Bitrix24 вернул пустой ID сделки при crm.deal.add.";
		Возврат Результат;
	КонецЕсли;
	
	Результат.СозданаНоваяСделка = Истина;
	Результат.IDСделкиB24 = IDСделки;
	
	Попытка
		PDF = Б24КП_ПечатьСервер.ТабличныйДокументВPDF(ПечатнаяФорма);
	Исключение
		Результат.ТекстОшибки = "Сделка создана (ID " + IDСделки
			+ "), но не удалось конвертировать КП в PDF: "
			+ КраткоеПредставлениеОшибки(ИнформацияОбОшибке());
		Возврат Результат;
	КонецПопытки;
	
	ИмяФайла = ПодготовитьИмяФайла(НазваниеПечатнойФормы) + ".pdf";
	ОтветФайла = ДобавитьФайлВТаймлайнСделки(ВебхукURL, IDСделки, ИмяФайла, PDF, НазваниеПечатнойФормы);
	
	Если ОтветФайла.Успех Тогда
		Результат.Успех = Истина;
	Иначе
		Результат.ТекстОшибки = "Сделка создана (ID " + IDСделки
			+ "), но не удалось добавить PDF в таймлайн: " + ОтветФайла.ТекстОшибки;
	КонецЕсли;
	
	Возврат Результат;
	
КонецФункции

Функция ПолучитьНастройкуПодключенияB24()
	
	Если Метаданные.ОбщиеМодули.Найти("Б24_К_ОбщегоНазначенияВызовСервераПовтИсп") = Неопределено Тогда
		Возврат Неопределено;
	КонецЕсли;
	
	МодульB24 = ОбщегоНазначения.ОбщийМодуль("Б24_К_ОбщегоНазначенияВызовСервераПовтИсп");
	
	Количество = МодульB24.ПолучитьКоличествоНастроекПодключения();

		Возврат Неопределено;
	КонецЕсли;

КонецФункции

Функция ПолучитьОбъектСвязиДляТаймЛайн(НастройкаПодключения, Объект)
	
	Если ЕстьСвязьОбъектаСБ24(НастройкаПодключения, Объект) Тогда
		Возврат Объект;
	КонецЕсли;
	
	Если Метаданные.ОбщиеМодули.Найти("Б24_К_ВыгрузкаДелСервер") = Неопределено Тогда
		Возврат Неопределено;
	КонецЕсли;
	
	МодульВыгрузка = ОбщегоНазначения.ОбщийМодуль("Б24_К_ВыгрузкаДелСервер");
	ДокументыОснования = МодульВыгрузка.ПолучитьДокументыОснования(Объект);
	
	Для Каждого ДокументОснование Из ДокументыОснования Цикл
		
		Если НЕ ЗначениеЗаполнено(ДокументОснование) Тогда
			Продолжить;
		КонецЕсли;
		
		Если ЕстьСвязьОбъектаСБ24(НастройкаПодключения, ДокументОснование) Тогда
			Возврат ДокументОснование;
		КонецЕсли;
		
	КонецЦикла;
	
	Возврат Неопределено;
	
КонецФункции

Функция ЕстьСвязьОбъектаСБ24(НастройкаПодключения, Объект)
	
	Если НЕ ЗначениеЗаполнено(Объект) Тогда
		Возврат Ложь;
	КонецЕсли;
	
	Если Метаданные.ОбщиеМодули.Найти("Б24_К_RestApiВызовСервера") = Неопределено Тогда
		Возврат Ложь;
	КонецЕсли;
	
	МодульRest = ОбщегоНазначения.ОбщийМодуль("Б24_К_RestApiВызовСервера");
	
	УстановитьПривилегированныйРежим(Истина);
	
	Запрос = Новый Запрос;
	Запрос.УстановитьПараметр("Портал", НастройкаПодключения.Портал);
	Запрос.УстановитьПараметр("ТипыДанных", МодульRest.ПолучитьТипыДанныхДел());
	Запрос.УстановитьПараметр("ТипОбъекта", Объект.Метаданные().ПолноеИмя());
	Запрос.УстановитьПараметр("ИдентификаторОбъекта", XMLСтрока(Объект));
	Запрос.Текст =
	"ВЫБРАТЬ
	|	Б24_К_ИдентификаторыДел.ДополнительныйИдентификатор КАК Идентификатор
	|ИЗ
	|	РегистрСведений.Б24_К_ИдентификаторыДел КАК Б24_К_ИдентификаторыДел
	|ГДЕ
	|	Б24_К_ИдентификаторыДел.Портал = &Портал
	|	И Б24_К_ИдентификаторыДел.ТипДанных В(&ТипыДанных)
	|	И Б24_К_ИдентификаторыДел.ТипОбъекта = &ТипОбъекта
	|	И Б24_К_ИдентификаторыДел.ИдентификаторОбъекта = &ИдентификаторОбъекта";
	
	Возврат НЕ Запрос.Выполнить().Пустой();
	
КонецФункции

	Если НЕ ЗначениеЗаполнено(Объект) Тогда
		Возврат "";
	КонецЕсли;
	
	МодульRest = ОбщегоНазначения.ОбщийМодуль("Б24_К_RestApiВызовСервера");
	
	УстановитьПривилегированныйРежим(Истина);

	Запрос.УстановитьПараметр("ТипОбъекта", Объект.Метаданные().ПолноеИмя());
	Запрос.УстановитьПараметр("ИдентификаторОбъекта", XMLСтрока(Объект));
	Запрос.Текст =
	"ВЫБРАТЬ ПЕРВЫЕ 1
	|	Б24_К_ИдентификаторыДел.ДополнительныйИдентификатор КАК Идентификатор

	|	Б24_К_ИдентификаторыДел.Портал = &Портал

	|	И Б24_К_ИдентификаторыДел.ИдентификаторОбъекта = &ИдентификаторОбъекта";
	
	Выборка = Запрос.Выполнить().Выбрать();
	Если Выборка.Следующий() Тогда
		Возврат СокрЛП(Строка(Выборка.Идентификатор));
	КонецЕсли;
	
	Возврат "";

Функция ДобавитьФайлВТаймлайнСделки(ВебхукURL, IDСделки, ИмяФайла, ДвоичныеДанныеPDF, Комментарий)
	
	Результат = Новый Структура("Успех, ТекстОшибки", Ложь, "");
	
	ФайлДляB24 = Новый Массив;
	ФайлДляB24.Добавить(ИмяФайла);
	ФайлДляB24.Добавить(Base64Строка(ДвоичныеДанныеPDF));
	
	Файлы = Новый Массив;
	Файлы.Добавить(ФайлДляB24);
	
	Поля = Новый Структура;
	Поля.Вставить("ENTITY_ID", IDСделки);
	Поля.Вставить("ENTITY_TYPE", "deal");
	Поля.Вставить("COMMENT", ?(ЗначениеЗаполнено(Комментарий), Комментарий, "Коммерческое предложение из 1С"));
	Поля.Вставить("FILES", Файлы);
	
	Ответ = ВызватьRestМетод(ВебхукURL, "crm.timeline.comment.add", Новый Структура("fields", Поля));
	
	Если Ответ.Успех Тогда
		Результат.Успех = Истина;
	Иначе
		Результат.ТекстОшибки = Ответ.ТекстОшибки;
	КонецЕсли;
	
	Возврат Результат;
	
КонецФункции

Функция ВызватьRestМетод(ВебхукURL, Метод, ПараметрыЗапроса)
	
	Результат = Новый Структура("Успех, Результат, ТекстОшибки", Ложь, Неопределено, "");
	
	БазовыйURL = НормализоватьURLВебхука(ВебхукURL);
	ПолныйURL = БазовыйURL + Метод + ".json";
	
	СтруктураURI = СтруктураURI(ПолныйURL);
	
	ЗащищенноеСоединение = Неопределено;
	Если НРег(СтруктураURI.Схема) = "https" Тогда
		ЗащищенноеСоединение = Новый ЗащищенноеСоединениеOpenSSL(Неопределено, Неопределено);
	КонецЕсли;
	
	Таймаут = 60;
	HTTPСоединение = Новый HTTPСоединение(
		СтруктураURI.Хост,
		СтруктураURI.Порт,
		,
		,
		,
		Таймаут,
		ЗащищенноеСоединение);
	
	ТелоОбъекта = Новый Структура;
	Если ТипЗнч(ПараметрыЗапроса) = Тип("Структура") Тогда
		Для Каждого КлючЗначение Из ПараметрыЗапроса Цикл
			ТелоОбъекта.Вставить(КлючЗначение.Ключ, КлючЗначение.Значение);
		КонецЦикла;
	КонецЕсли;
	
	ЗаписьJSON = Новый ЗаписьJSON;
	ЗаписьJSON.УстановитьСтроку();
	ЗаписатьJSON(ЗаписьJSON, ТелоОбъекта);
	ТелоJSON = ЗаписьJSON.Закрыть();
	
	HTTPЗапрос = Новый HTTPЗапрос(СтруктураURI.ПутьНаСервере);
	HTTPЗапрос.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");
	HTTPЗапрос.УстановитьТелоИзСтроки(ТелоJSON, КодировкаТекста.UTF8, ИспользованиеByteOrderMark.НеИспользовать);
	
	Попытка
		HTTPОтвет = HTTPСоединение.ОтправитьДляОбработки(HTTPЗапрос);
	Исключение
		Результат.ТекстОшибки = "HTTP-ошибка при вызове " + Метод + ": "
			+ КраткоеПредставлениеОшибки(ИнформацияОбОшибке());
		ЗаписьЖурналаРегистрации(
			"Б24КП.RestAPI",
			УровеньЖурналаРегистрации.Ошибка,
			,
			,
			Результат.ТекстОшибки + Символы.ПС + ПодробноеПредставлениеОшибки(ИнформацияОбОшибке()));
		Возврат Результат;
	КонецПопытки;
	
	КодСостояния = HTTPОтвет.КодСостояния;
	ТелоОтвета = HTTPОтвет.ПолучитьТелоКакСтроку(КодировкаТекста.UTF8);
	
	Если КодСостояния < 200 Или КодСостояния >= 300 Тогда
		Результат.ТекстОшибки = СтрШаблон("Bitrix24 HTTP %1 при %2: %3", КодСостояния, Метод, Лев(ТелоОтвета, 500));
		ЗаписьЖурналаРегистрации("Б24КП.RestAPI", УровеньЖурналаРегистрации.Ошибка, , , Результат.ТекстОшибки);
		Возврат Результат;
	КонецЕсли;
	
	Попытка
		ЧтениеJSON = Новый ЧтениеJSON;
		ЧтениеJSON.УстановитьСтроку(ТелоОтвета);
		ОтветОбъект = ПрочитатьJSON(ЧтениеJSON);
		ЧтениеJSON.Закрыть();
	Исключение
		Результат.ТекстОшибки = "Некорректный JSON от Bitrix24 (" + Метод + "): " + Лев(ТелоОтвета, 300);
		Возврат Результат;
	КонецПопытки;
	
	Если ТипЗнч(ОтветОбъект) = Тип("Структура") Тогда
		
		Если ОтветОбъект.Свойство("error") И ЗначениеЗаполнено(ОтветОбъект.error) Тогда
			Описание = "";
			Если ОтветОбъект.Свойство("error_description") Тогда
				Описание = Строка(ОтветОбъект.error_description);
			КонецЕсли;
			Результат.ТекстОшибки = СтрШаблон("%1: %2 (%3)", Метод, ОтветОбъект.error, Описание);
			ЗаписьЖурналаРегистрации("Б24КП.RestAPI", УровеньЖурналаРегистрации.Ошибка, , , Результат.ТекстОшибки);
			Возврат Результат;
		КонецЕсли;
		
		Если ОтветОбъект.Свойство("result") Тогда
			Результат.Успех = Истина;

		КонецЕсли;
		
	КонецЕсли;
	
	Результат.ТекстОшибки = "Неожиданный ответ Bitrix24 (" + Метод + "): " + Лев(ТелоОтвета, 300);
	Возврат Результат;

Функция СтруктураURI(Знач URL)
	
	Результат = Новый Структура("Схема, Хост, Порт, ПутьНаСервере", "https", "", 443, "/");
	
	URL = СокрЛП(URL);
	ПозицияСхемы = СтрНайти(URL, "://");
	Если ПозицияСхемы > 0 Тогда
		Результат.Схема = НРег(Лев(URL, ПозицияСхемы - 1));
		URL = Сред(URL, ПозицияСхемы + 3);
	КонецЕсли;
	
	ПозицияПути = СтрНайти(URL, "/");
	ХостПорт = ?(ПозицияПути = 0, URL, Лев(URL, ПозицияПути - 1));
	Результат.ПутьНаСервере = ?(ПозицияПути = 0, "/", Сред(URL, ПозицияПути));
	
	ПозицияПорта = СтрНайти(ХостПорт, ":");
	Если ПозицияПорта > 0 Тогда
		Результат.Хост = Лев(ХостПорт, ПозицияПорта - 1);
		Результат.Порт = Число(Сред(ХостПорт, ПозицияПорта + 1));
	Иначе
		Результат.Хост = ХостПорт;
		Результат.Порт = ?(Результат.Схема = "https", 443, 80);
	КонецЕсли;
	
	Возврат Результат;
