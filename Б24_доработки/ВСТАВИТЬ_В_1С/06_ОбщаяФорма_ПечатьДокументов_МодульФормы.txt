&НаКлиенте
&После("ПриСозданииНаСервере")
Процедура Б24_ПослеПриСозданииНаСервере(Отказ, СтандартнаяОбработка)
	Б24_ПрименитьБлокировкуПечатиНаКлиенте();
КонецПроцедуры

&НаКлиенте
&Вместо("Печать")
Процедура Б24_ВместоПечать(Команда)

	Если НЕ Б24_ПроверитьДатуПередПечатьюСчетаНаКлиенте() Тогда
		Возврат;
	КонецЕсли;

	ПродолжитьВызов(Команда);

КонецПроцедуры

&НаКлиенте
Процедура Б24_ПрименитьБлокировкуПечатиНаКлиенте()

	Результат = Б24_ОценитьПроверкуДатыНаСервере();

	Если НЕ Результат.Проверять Тогда
		Возврат;
	КонецЕсли;

	Если ПустаяСтрока(Результат.ТекстОшибки) Тогда
		Возврат;
	КонецЕсли;

	ПоказатьПредупреждение(, Результат.ТекстОшибки);
	Б24_УстановитьДоступностьПечати(Ложь);

КонецПроцедуры

&НаКлиенте
Процедура Б24_УстановитьДоступностьПечати(Доступность)

	ИменаКоманд = Новый Массив;
	ИменаКоманд.Добавить("Печать");
	ИменаКоманд.Добавить("ПечатьДокументов");

	Для Каждого ИмяКоманды Из ИменаКоманд Цикл

		Элемент = Элементы.Найти(ИмяКоманды);

		Если Элемент <> Неопределено Тогда
			Элемент.Доступность = Доступность;
		КонецЕсли;

	КонецЦикла;

КонецПроцедуры

&НаКлиенте
Функция Б24_ПроверитьДатуПередПечатьюСчетаНаКлиенте()

	Результат = Б24_ОценитьПроверкуДатыНаСервере();

	Если НЕ Результат.Проверять Тогда
		Возврат Истина;
	КонецЕсли;

	Если ПустаяСтрока(Результат.ТекстОшибки) Тогда
		Возврат Истина;
	КонецЕсли;

	ПоказатьПредупреждение(, Результат.ТекстОшибки);
	Возврат Ложь;

КонецФункции

&НаСервере
Функция Б24_ОценитьПроверкуДатыНаСервере()

	Результат = Новый Структура("Проверять, ТекстОшибки", Ложь, "");
	ЗаказСсылка = Б24_ЗаказПоОбъектамПечати();

	Если Б24_ЕстьФаксимилеСчетВНастройках() Тогда
		Результат.Проверять = Истина;
	ИначеЕсли Б24_ЕстьФаксимилеСчетВПараметрах() Тогда
		Результат.Проверять = Истина;
	Иначе
		Возврат Результат;
	КонецЕсли;

	Результат.ТекстОшибки = Б24_СчетСервер.ТекстОшибкиДатыПередСчетомСФаксимиле(ЗаказСсылка);

	Возврат Результат;

КонецФункции

&НаСервере
Функция Б24_ЕстьФаксимилеСчетВНастройках()

	Если НастройкиПечатныхФорм = Неопределено Тогда
		Возврат Ложь;
	КонецЕсли;

	Для Каждого СтрокаНастройки Из НастройкиПечатныхФорм Цикл

		Если НЕ Б24_СтрокаВыбранаДляПечати(СтрокаНастройки) Тогда
			Продолжить;
		КонецЕсли;

		Если Б24_СтрокаНастройкиЭтоФаксимилеСчет(СтрокаНастройки) Тогда
			Возврат Истина;
		КонецЕсли;

	КонецЦикла;

	Возврат Ложь;

КонецФункции

&НаСервере
Функция Б24_ЕстьФаксимилеСчетВПараметрах()

	Если Параметры.Свойство("ИменаМакетов") Тогда

		Если Б24_СчетСервер.ЭтоПечатьСчетаСФаксимиле(Параметры.ИменаМакетов, Параметры.ИменаМакетов) Тогда
			Возврат Истина;
		КонецЕсли;

	КонецЕсли;

	Если НЕ Параметры.Свойство("КоллекцияПечатныхФорм") Тогда
		Возврат Ложь;
	КонецЕсли;

	Коллекция = Параметры.КоллекцияПечатныхФорм;

	Если Коллекция = Неопределено Тогда
		Возврат Ложь;
	КонецЕсли;

	Для Каждого ПечатнаяФорма Из Коллекция Цикл

		ИмяМакета = Б24_ЗначениеКолонкиСтроки(ПечатнаяФорма, "ИмяМакета,Идентификатор,Имя,ИмяМакета");
		Название = Б24_ЗначениеКолонкиСтроки(ПечатнаяФорма, "Синоним,Название,Представление");

		Если Б24_СчетСервер.ЭтоПечатьСчетаСФаксимиле(ИмяМакета, Название) Тогда
			Возврат Истина;
		КонецЕсли;

	КонецЦикла;

	Возврат Ложь;

КонецФункции

&НаСервере
Функция Б24_СтрокаНастройкиЭтоФаксимилеСчет(СтрокаНастройки)

	ИмяМакета = Б24_ЗначениеКолонкиСтроки(СтрокаНастройки, "ИмяМакета,Идентификатор,Имя,ИмяРеквизита");
	Название = Б24_ЗначениеКолонкиСтроки(СтрокаНастройки, "Название,Представление,Синоним");

	Возврат Б24_СчетСервер.ЭтоПечатьСчетаСФаксимиле(ИмяМакета, Название);

КонецФункции

&НаСервере
Функция Б24_СтрокаВыбранаДляПечати(СтрокаНастройки)

	ИменаКолонок = "Печать,Печатать,Выбран,ВыводитьНаПечать,Пометка";

	Для Каждого ИмяКолонки Из СтрРазделить(ИменаКолонок, ",", Ложь) Цикл

		Если НЕ ОбщегоНазначения.ЕстьРеквизитИлиСвойствоОбъекта(СтрокаНастройки, ИмяКолонки) Тогда
			Продолжить;
		КонецЕсли;

		Возврат СтрокаНастройки[ИмяКолонки];

	КонецЦикла;

	Возврат Истина;

КонецФункции

&НаСервере
Функция Б24_ЗначениеКолонкиСтроки(СтрокаНастройки, ИменаКолонок)

	Для Каждого ИмяКолонки Из СтрРазделить(ИменаКолонок, ",", Ложь) Цикл

		Если НЕ ОбщегоНазначения.ЕстьРеквизитИлиСвойствоОбъекта(СтрокаНастройки, ИмяКолонки) Тогда
			Продолжить;
		КонецЕсли;

		Значение = СтрокаНастройки[ИмяКолонки];

		Если ЗначениеЗаполнено(Значение) Тогда
			Возврат Значение;
		КонецЕсли;

	КонецЦикла;

	Возврат "";

КонецФункции

&НаСервере
Функция Б24_ЗаказПоОбъектамПечати()

	Если ОбъектыПечати <> Неопределено И ОбъектыПечати.Количество() > 0 Тогда

		Элемент = ОбъектыПечати[0];

		Если ТипЗнч(ОбъектыПечати) = Тип("СписокЗначений") Тогда
			ОбъектПечати = Элемент.Значение;
		ИначеЕсли ТипЗнч(Элемент) = Тип("Структура") И Элемент.Свойство("Значение") Тогда
			ОбъектПечати = Элемент.Значение;
		Иначе
			ОбъектПечати = Элемент;
		КонецЕсли;

		Заказ = Б24_СчетСервер.ЗаказКлиентаПоОбъектуПечати(ОбъектПечати);

		Если ЗначениеЗаполнено(Заказ) Тогда
			Возврат Заказ;
		КонецЕсли;

	КонецЕсли;

	Если Параметры.Свойство("ПараметрКоманды") Тогда

		ПараметрКоманды = Параметры.ПараметрКоманды;

		Если ТипЗнч(ПараметрКоманды) = Тип("Массив") Тогда

			Для Каждого ОбъектПечати Из ПараметрКоманды Цикл

				Заказ = Б24_СчетСервер.ЗаказКлиентаПоОбъектуПечати(ОбъектПечати);

				Если ЗначениеЗаполнено(Заказ) Тогда
					Возврат Заказ;
				КонецЕсли;

			КонецЦикла;

		Иначе

			Заказ = Б24_СчетСервер.ЗаказКлиентаПоОбъектуПечати(ПараметрКоманды);

			Если ЗначениеЗаполнено(Заказ) Тогда
				Возврат Заказ;
			КонецЕсли;

		КонецЕсли;

	КонецЕсли;

	Возврат Документы.ЗаказКлиента.ПустаяСсылка();

КонецФункции
