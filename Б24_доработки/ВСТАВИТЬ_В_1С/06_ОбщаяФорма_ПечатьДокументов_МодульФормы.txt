#Область Б24_ПроверкаДатыПередСчетомСФаксимиле

&НаКлиенте
&Вместо("Печать")
Процедура Б24_ВместоПечать(Команда)

	Если НЕ Б24_ПроверитьДатуПередПечатьюСчетаНаКлиенте() Тогда
		Возврат;
	КонецЕсли;

	ПродолжитьВызов(Команда);

КонецПроцедуры

&НаКлиенте
Функция Б24_ПроверитьДатуПередПечатьюСчетаНаКлиенте()

	Результат = Б24_ПроверитьДатуПередПечатьюСчетаНаСервере();

	Если НЕ Результат.Проверять Тогда
		Возврат Истина;
	КонецЕсли;

	Если ПустаяСтрока(Результат.ТекстОшибки) Тогда
		Возврат Истина;
	КонецЕсли;

	ПоказатьПредупреждение(, Результат.ТекстОшибки);
	Возврат Ложь;

КонецФункции

&НаСервере
Функция Б24_ПроверитьДатуПередПечатьюСчетаНаСервере()

	Результат = Новый Структура("Проверять, ТекстОшибки", Ложь, "");

	Если НастройкиПечатныхФорм = Неопределено Тогда
		Возврат Результат;
	КонецЕсли;

	НужнаПроверка = Ложь;
	ЗаказСсылка = Документы.ЗаказКлиента.ПустаяСсылка();

	Для Каждого СтрокаНастройки Из НастройкиПечатныхФорм Цикл

		Если НЕ СтрокаНастройки.Печать Тогда
			Продолжить;
		КонецЕсли;

		Если НЕ Б24_СчетСервер.ЭтоПечатьСчетаСФаксимиле(СтрокаНастройки.ИмяМакета, СтрокаНастройки.Название) Тогда
			Продолжить;
		КонецЕсли;

		НужнаПроверка = Истина;

		Если ОбъектыПечати.Количество() > 0 Тогда
			ЗаказСсылка = Б24_СчетСервер.ЗаказКлиентаПоОбъектуПечати(ОбъектыПечати[0].Значение);
		КонецЕсли;

		Прервать;

	КонецЦикла;

	Если НЕ НужнаПроверка Тогда
		Возврат Результат;
	КонецЕсли;

	Результат.Проверять = Истина;
	Результат.ТекстОшибки = Б24_СчетСервер.ТекстОшибкиДатыПередСчетомСФаксимиле(ЗаказСсылка);

	Возврат Результат;

КонецФункции

#КонецОбласти
