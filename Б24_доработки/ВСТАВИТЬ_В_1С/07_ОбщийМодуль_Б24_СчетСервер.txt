#Область ПрограммныйИнтерфейс

// Пустая строка = проверка пройдена, иначе текст для пользователя.
Функция ТекстОшибкиДатыПередСчетомСФаксимиле(ЗаказСсылка) Экспорт

	Если НЕ ЗначениеЗаполнено(ЗаказСсылка) Тогда
		Возврат "";
	КонецЕсли;

	Если ЗначениеЗаполнено(Б24_ДатаОплатыЗаказа(ЗаказСсылка)) Тогда
		Возврат "";
	КонецЕсли;

	Возврат "Заполните «Дату оплаты» на заказе клиента перед печатью счёта с факсимиле.";

КонецФункции

// Заказ по объекту печати (заказ или счёт на оплату).
Функция ЗаказКлиентаПоОбъектуПечати(ОбъектПечати) Экспорт

	Если НЕ ЗначениеЗаполнено(ОбъектПечати) Тогда
		Возврат Документы.ЗаказКлиента.ПустаяСсылка();
	КонецЕсли;

	Если ТипЗнч(ОбъектПечати) = Тип("ДокументСсылка.ЗаказКлиента") Тогда
		Возврат ОбъектПечати;
	КонецЕсли;

	Если ТипЗнч(ОбъектПечати) = Тип("ДокументСсылка.СчетНаОплатуКлиенту") Тогда

		Попытка
			Заказ = ОбщегоНазначения.ЗначениеРеквизитаОбъекта(ОбъектПечати, "ДокументОснование");
			Если ТипЗнч(Заказ) = Тип("ДокументСсылка.ЗаказКлиента") Тогда
				Возврат Заказ;
			КонецЕсли;
		Исключение
		КонецПопытки;

	КонецЕсли;

	Возврат Документы.ЗаказКлиента.ПустаяСсылка();

КонецФункции

// Выбрана печать счёта с факсимиле (по имени/названию макета).
Функция ЭтоПечатьСчетаСФаксимиле(ИмяМакета, НазваниеПечатнойФормы) Экспорт

	Текст = НРег(СокрЛП(Строка(ИмяМакета) + " " + Строка(НазваниеПечатнойФормы)));

	// Достаточно слова «факсимил» — в меню заказа это «Счёт с факсимиле».
	Если СтрНайти(Текст, "факсимил") > 0 Тогда
		Возврат Истина;
	КонецЕсли;

	Если СтрНайти(Текст, "facsimile") > 0 Тогда
		Возврат Истина;
	КонецЕсли;

	Возврат Ложь;

КонецФункции

#КонецОбласти

#Область СлужебныеПроцедурыИФункции

Функция Б24_ДатаОплатыЗаказа(ЗаказСсылка)

	Если Метаданные.ОбщиеМодули.Найти("Б24_ОплатаСервер") <> Неопределено Тогда
		Модуль = ОбщегоНазначения.ОбщийМодуль("Б24_ОплатаСервер");
		Возврат Модуль.ДатаФактОплатыПоЗаказу(ЗаказСсылка);
	КонецЕсли;

	Попытка
		Возврат ОбщегоНазначения.ЗначениеРеквизитаОбъекта(ЗаказСсылка, "Б24_ДатаФактОплаты");
	Исключение
		Возврат '00010101';
	КонецПопытки;

КонецФункции

#КонецОбласти
