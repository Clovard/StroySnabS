&После("ПриОткрытии")
Процедура Б24_ПослеПриОткрытии(Отказ)

	КнопкаКП = Элементы.Найти("Б24_СформироватьКП");

	Если КнопкаКП <> Неопределено Тогда
		КнопкаКП.УстановитьДействие("Нажатие", "Б24_СформироватьКП");
	КонецЕсли;

КонецПроцедуры

&НаКлиенте
Процедура Б24_СформироватьКП(Команда)

	Если НЕ ЗначениеЗаполнено(Объект.Ссылка) Или Объект.Ссылка.Пустая() Тогда
		ПоказатьПредупреждение(, "Сначала запишите заказ клиента.");
		Возврат;
	КонецЕсли;

	Если Модифицированность Тогда
		ПоказатьПредупреждение(, "Заказ изменён. Запишите документ перед формированием КП.");
		Возврат;
	КонецЕсли;

	Состояние("Формирование КП и отправка в Bitrix24...");

	Результат = Б24КП_ВызовСервера.СформироватьКПиОтправитьВBitrix24(Объект.Ссылка);

	Если Результат.PDFСформирован И Результат.ПечатнаяФорма <> Неопределено Тогда
		Результат.ПечатнаяФорма.Показать(Результат.НазваниеПечатнойФормы);
	КонецЕсли;

	Если Результат.PDFСформирован Тогда
		Попытка
			Прочитать();
		Исключение
			ОбновитьОтображениеДанных();
		КонецПопытки;
	КонецЕсли;

	Если Результат.Успех Тогда
		ПоказатьОповещениеПользователя("Сформировать КП", , Результат.ТекстСообщения);
	ИначеЕсли ЗначениеЗаполнено(Результат.ТекстСообщения) Тогда
		ПоказатьПредупреждение(, Результат.ТекстСообщения);
	ИначеЕсли ЗначениеЗаполнено(Результат.ТекстОшибки) Тогда
		ПоказатьПредупреждение(, Результат.ТекстОшибки);
	Иначе
		ПоказатьПредупреждение(, "Не удалось сформировать КП.");
	КонецЕсли;

		Возврат;

	СтандартнаяОбработка = Ложь;

&НаКлиенте
Функция Б24_ЭтоФаксимилеКоманда(Команда, ДополнительныеПараметры)

	Части = Новый Массив;

	Если Команда <> Неопределено Тогда
		Попытка
			Части.Добавить(Строка(Команда.Имя));
		Исключение
		КонецПопытки;
		Попытка
			Части.Добавить(Строка(Команда.Заголовок));
		Исключение
		КонецПопытки;
	КонецЕсли;

	Если ДополнительныеПараметры <> Неопределено
		И ТипЗнч(ДополнительныеПараметры) = Тип("Структура")
		И ДополнительныеПараметры.Свойство("ИменаМакетов") Тогда
		Части.Добавить(Строка(ДополнительныеПараметры.ИменаМакетов));
	КонецЕсли;

	Текст = НРег(СтрСоединить(Части, " "));

	Возврат СтрНайти(Текст, "факсимил") > 0
		Или СтрНайти(Текст, "facsimile") > 0;
